113template <typename T> class ComponentArray : public IComponentArray {
114 public:
115 void InsertData(Entity entity, T component) {
116 assert(mEntityToIndexMap.find(entity) == mEntityToIndexMap.end() && "Component added to same entity more than once.");
117
118 // Put new entry at end and update the maps
119 size_t newIndex = mSize;
120 mEntityToIndexMap[entity] = newIndex;
121 mIndexToEntityMap[newIndex] = entity;
122 mComponentArray[newIndex] = component;
123 ++mSize;
124 }
125
126 void RemoveData(Entity entity) {
127 assert(mEntityToIndexMap.find(entity) != mEntityToIndexMap.end() && "Removing non-existent component.");
128
129 // Copy element at end into deleted element's place to maintain density
130 size_t indexOfRemovedEntity = mEntityToIndexMap[entity];
131 size_t indexOfLastElement = mSize - 1;
132 mComponentArray[indexOfRemovedEntity] = mComponentArray[indexOfLastElement];
133
134 // Update map to point to moved spot
135 Entity entityOfLastElement = mIndexToEntityMap[indexOfLastElement];
136 mEntityToIndexMap[entityOfLastElement] = indexOfRemovedEntity;
137 mIndexToEntityMap[indexOfRemovedEntity] = entityOfLastElement;
138
139 mEntityToIndexMap.erase(entity);
140 mIndexToEntityMap.erase(indexOfLastElement);
141
142 --mSize;
143 }
144
145 T &GetData(Entity entity) {
146 assert(mEntityToIndexMap.find(entity) != mEntityToIndexMap.end() && "Retrieving non-existent component.");
147
148 // Return a reference to the entity's component
149 return mComponentArray[mEntityToIndexMap[entity]];
150 }
151
152 void EntityDestroyed(Entity entity) override {
153 if (mEntityToIndexMap.find(entity) != mEntityToIndexMap.end()) {
154 // Remove the entity's component if it existed
155 RemoveData(entity);
156 }
157 }
158
159 void Clear() override {
160 mEntityToIndexMap.clear();
161 mIndexToEntityMap.clear();
162 mSize = 0;
163 }
164
165 private:
166 // The packed array of components (of generic type T),
167 // set to a specified maximum amount, matching the maximum number
168 // of entities allowed to exist simultaneously, so that each entity
169 // has a unique spot.
170 std::array<T, MAX_ENTITIES> mComponentArray;
171
172 // Map from an entity ID to an array index.
173 std::unordered_map<Entity, size_t> mEntityToIndexMap;
174
175 // Map from an array index to an entity ID.
176 std::unordered_map<size_t, Entity> mIndexToEntityMap;
177
178 // Total size of valid entries in the array.
179 size_t mSize;
180};

297 void EntitySignatureChanged(Entity entity, Signature entitySignature) {
298 // Notify each system that an entity's signature changed
299 for (auto const &pair : mSystems) {
300 auto const &type = pair.first;
301 auto const &system = pair.second;
302 auto const &systemSignature = mSignatures[type];
303
304 // Entity signature matches system signature - insert into set
305 if ((entitySignature & systemSignature) == systemSignature) {
306 system->mEntities.insert(entity);
307 }
308 // Entity signature does not match system signature - erase from set
309 else {
310 system->mEntities.erase(entity);
311 }
312 }
313 }

317 auto const &system = pair.second;
318 system->mEntities.clear();
319 }
320 }
